13.TO DISPLAY CUSTOMERS WHOSE AGE IS ABOVE THE AVERAGE AND WHO LIVE IN A CITY WITH MORE THAN 2 CUSTOMERS
 
SQL>  WITH AvgAge AS (
  2       SELECT AVG(AGE) AS avg_age
  3       FROM USERS
  4     ),
  5     CityCount AS (
  6       SELECT ADDRESS_CITY, COUNT(*) AS city_count
  7       FROM USERS
  8       GROUP BY ADDRESS_CITY
  9     )
 10     SELECT C1.NAME, C1.AGE, C2.ADDRESS_CITY, C2.city_count
 11     FROM USERS C1
 12     JOIN CityCount C2 ON C1.ADDRESS_CITY = C2.ADDRESS_CITY
 13     WHERE C1.AGE < (SELECT avg_age FROM AvgAge)
